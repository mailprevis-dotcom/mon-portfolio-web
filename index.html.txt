    <script>
        // Ce script gère l'affichage du statut après l'envoi du formulaire Formspree
        const form = document.getElementById('contact-form');
        const status = document.getElementById('form-status');

        form.addEventListener("submit", async function(event) {
            event.preventDefault();
            const data = new FormData(event.target);
            
            try {
                const response = await fetch(event.target.action, {
                    method: form.method,
                    body: data,
                    headers: {
                        'Accept': 'application/json'
                    }
                });

                if (response.ok) {
                    status.textContent = "Merci ! Votre message a été envoyé avec succès.";
                    status.style.color = "green";
                    form.reset();
                } else {
                    status.textContent = "Une erreur s'est produite lors de l'envoi du message.";
                    status.style.color = "red";
                }
                status.style.display = "block";

            } catch (error) {
                console.error('Erreur de soumission du formulaire :', error);
                status.textContent = "Problème de connexion. Veuillez réessayer.";
                status.style.color = "red";
                status.style.display = "block";
            }
        });
    </script>
